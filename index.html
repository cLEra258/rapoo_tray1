connectBtn.addEventListener('click', async () => { console.log("按钮已点击，准备请求设备..."); if (!navigator.hid) { alert('当前环境不支持 WebHID'); return; } try { // 请求设备 console.log("正在调用 navigator.hid.requestDevice..."); const devices = await navigator.hid.requestDevice({ filters: [{ vendorId: 0x24AE }] }); console.log("requestDevice 返回结果:", devices); if (devices.length === 0) { statusEl.textContent = '未选择设备或已取消'; return; } // 遍历设备并打开 for (const dev of devices) { console.log(`尝试打开设备: ${dev.productName}, opened: ${dev.opened}`); if (!dev.opened) { await dev.open(); } dev.addEventListener('inputreport', onInputReport); console.log("设备打开成功并已绑定监听器"); } statusEl.textContent = `已连接：${devices[0].productName || 'Rapoo'}，请在托盘中查看电量。`; statusEl.style.color = '#059669'; connectBtn.disabled = true; window.electronAPI.updateBattery({ battery: -1, iconBase64: createIconBase64(-1) }); } catch (err) { console.error("连接过程发生错误:", err); statusEl.textContent = '连接失败：' + err.message; statusEl.style.color = '#dc2626'; } });
